      document.getElementById('btn-mode-full').className = `px-3 py-1.5 text-xs font-semibold rounded-md transition ${mode === 'full' ? 'bg-emerald-500 text-slate-950' : 'bg-slate-800 text-slate-300'}`;

      document.getElementById('btn-mode-first').className = `px-3 py-1.5 text-xs font-semibold rounded-md transition ${mode === 'first' ? 'bg-emerald-500 text-slate-950' : 'bg-slate-800 text-slate-300'}`;

      document.getElementById('btn-mode-hidden').className = `px-3 py-1.5 text-xs font-semibold rounded-md transition ${mode === 'hidden' ? 'bg-emerald-500 text-slate-950' : 'bg-slate-800 text-slate-300'}`;

      renderCard();

    }


    function toggleCardMask() {

      isCardMasked = !isCardMasked;

      renderCard();

    }


    function renderCard() {

      const item = SCRIPT_DATA[currentCardIdx];

      document.getElementById('card-stage-badge').innerText = item.stage;

      document.getElementById('card-counter').innerText = `${currentCardIdx + 1} / ${SCRIPT_DATA.length}`;

      document.getElementById('card-anchor').innerText = `Anchor: ${item.anchor}`;


      const cardTextEl = document.getElementById('card-text');


      if (isCardMasked) {

        cardTextEl.innerHTML = `<span class="bg-slate-800 px-4 py-2 rounded-lg text-slate-500 font-mono italic">[ Tap card to reveal text ]</span>`;

      } else if (cardErasureMode === 'first') {

        cardTextEl.innerHTML = `<span class="font-mono tracking-wider text-emerald-300 text-2xl font-bold">${item.firstCode}</span>`;

      } else if (cardErasureMode === 'hidden') {

        // Mask alternate key words

        const masked = item.full.replace(/\b([a-zA-Z]{4,})\b/g, '____');

        cardTextEl.innerText = masked;

      } else {

        cardTextEl.innerText = item.full;

      }


      // Update Prev / Next Buttons

      document.getElementById('btn-card-prev').disabled = currentCardIdx === 0;

      document.getElementById('btn-card-next').disabled = currentCardIdx === SCRIPT_DATA.length - 1;


      // Update Jump Buttons

      SCRIPT_DATA.forEach((_, idx) => {

        const btn = document.getElementById(`jump-btn-${idx}`);

        if (btn) {

          if (idx === currentCardIdx) {

            btn.className = "p-2 rounded-lg border text-xs font-mono font-bold transition flex flex-col items-center justify-center gap-1 bg-emerald-500 text-slate-950 border-emerald-400 shadow-md shadow-emerald-500/20";

          } else if (masteredCards.has(idx)) {

            btn.className = "p-2 rounded-lg border text-xs font-mono font-bold transition flex flex-col items-center justify-center gap-1 bg-emerald-950/60 text-emerald-400 border-emerald-800";

          } else {

            btn.className = "p-2 rounded-lg border text-xs font-mono font-bold transition flex flex-col items-center justify-center gap-1 bg-slate-900 text-slate-400 border-slate-800 hover:border-slate-700";

          }

        }

      });

    }


    function navCard(direction) {

      currentCardIdx = Math.max(0, Math.min(SCRIPT_DATA.length - 1, currentCardIdx + direction));

      isCardMasked = false;

      renderCard();

    }


    function markCardMastered() {

      masteredCards.add(currentCardIdx);

      userStreak += 1;

      updateMasteryStats();

      if (currentCardIdx < SCRIPT_DATA.length - 1) {

        navCard(1);

      } else {

        renderCard();

      }

    }


    // --- MODE 2: FILL-IN-THE-BLANKS ---

    function renderFillBlanks() {

      const container = document.getElementById('fill-blanks-container');

      container.innerHTML = '';


      SCRIPT_DATA.forEach((item, index) => {

        const card = document.createElement('div');

        card.className = "glass-panel p-5 rounded-xl border border-slate-800 space-y-3";


        // Generate template with input boxes

        let templateHtml = item.blankTemplate;

        item.blanks.forEach(word => {

          const inputHtml = `<input type="text" data-answer="${word.toLowerCase()}" class="fill-input inline-block w-28 px-2 py-1 mx-1 text-sm bg-slate-950 border border-slate-700 rounded text-emerald-400 focus:outline-none focus:border-emerald-500 font-mono text-center" placeholder="..." />`;

          templateHtml = templateHtml.replace(`[${word}]`, inputHtml);

        });


        card.innerHTML = `

          <div class="flex items-center justify-between text-xs text-slate-400 border-b border-slate-800 pb-2">

            <span class="font-semibold text-emerald-400">Step ${index + 1}: ${item.anchor}</span>

            <span>${item.stage}</span>

          </div>

          <div class="text-base font-medium text-slate-200 leading-relaxed pt-1">

            ${templateHtml}

          </div>

          <div id="fill-feedback-${index}" class="text-xs font-mono hidden pt-1"></div>

        `;

        container.appendChild(card);

      });

    }


    function checkAllFillBlanks() {

      let correct = 0;

      let total = 0;


      const inputs = document.querySelectorAll('.fill-input');

      inputs.forEach(input => {

        total++;

        const val = input.value.trim().toLowerCase();

        const expected = input.getAttribute('data-answer');


        if (val === expected) {

          correct++;

          input.classList.remove('border-slate-700', 'border-rose-500');

          input.classList.add('border-emerald-500', 'bg-emerald-950/40');

        } else {

          input.classList.remove('border-slate-700', 'border-emerald-500');

          input.classList.add('border-rose-500', 'bg-rose-950/40');

        }

      });


      if (correct === total) {

        userStreak += 2;

        updateMasteryStats();

      }

    }


    // --- MODE 3: SEQUENCE ORDERING ---

    function initSequence() {

      shuffleSequence();

    }


    function shuffleSequence() {

      sequenceItems = [...SCRIPT_DATA].sort(() => Math.random() - 0.5);

      renderSequence();

      document.getElementById('sequence-alert').classList.add('hidden');

    }


    function renderSequence() {

      const list = document.getElementById('sequence-list');

      list.innerHTML = '';


      sequenceItems.forEach((item, idx) => {

        const itemEl = document.createElement('div');

        itemEl.className = "glass-panel p-4 rounded-xl border border-slate-800 flex items-center justify-between gap-3 cursor-pointer hover:border-slate-700 transition";

        itemEl.innerHTML = `

          <div class="flex items-center gap-3">

            <span class="w-7 h-7 rounded-lg bg-slate-800 text-slate-300 font-mono text-xs flex items-center justify-center font-bold">

              ${idx + 1}

            </span>

            <p class="text-sm font-medium text-slate-200">${item.full}</p>

          </div>

          <div class="flex items-center gap-1">

            <button onclick="moveSequence(${idx}, -1)" class="p-2 rounded-lg bg-slate-800 hover:bg-slate-700 text-slate-300 text-xs" ${idx === 0 ? 'disabled class="opacity-30"' : ''}>

              <i class="fa-solid fa-arrow-up"></i>

            </button>

            <button onclick="moveSequence(${idx}, 1)" class="p-2 rounded-lg bg-slate-800 hover:bg-slate-700 text-slate-300 text-xs" ${idx === sequenceItems.length - 1 ? 'disabled class="opacity-30"' : ''}>

              <i class="fa-solid fa-arrow-down"></i>

            </button>

          </div>

        `;

        list.appendChild(itemEl);

      });

    }


    function moveSequence(idx, direction) {

      const targetIdx = idx + direction;

      if (targetIdx < 0 || targetIdx >= sequenceItems.length) return;

      const temp = sequenceItems[idx];

      sequenceItems[idx] = sequenceItems[targetIdx];

      sequenceItems[targetIdx] = temp;

      renderSequence();

    }


    function verifySequence() {

      let isCorrect = true;

      sequenceItems.forEach((item, idx) => {

        if (item.id !== idx + 1) isCorrect = false;

      });


      const alert = document.getElementById('sequence-alert');

      alert.classList.remove('hidden', 'bg-emerald-950/80', 'border-emerald-500/50', 'text-emerald-300', 'bg-rose-950/80', 'border-rose-500/50', 'text-rose-300');


      if (isCorrect) {

        alert.classList.add('bg-emerald-950/80', 'border-emerald-500/50', 'text-emerald-300');

        alert.innerHTML = `<span><i class="fa-solid fa-circle-check mr-2"></i><strong>Perfect Sequence!</strong> You have successfully mastered the 3-Stage coaching agenda flow.</span>`;

        userStreak += 3;

        updateMasteryStats();

      } else {

        alert.classList.add('bg-rose-950/80', 'border-rose-500/50', 'text-rose-300');

        alert.innerHTML = `<span><i class="fa-solid fa-circle-xmark mr-2"></i><strong>Incorrect Order.</strong> Remember: Stage 1 (Exploration) &rarr; Stage 2 (Target Outcome) &rarr; Stage 3 (Motivation).</span>`;

      }

    }

    function setupSpeechRecognition() {

      if ('webkitSpeechRecognition' in window || 'SpeechRecognition' in window) {

        const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;

        recognition = new SpeechRecognition();

        recognition.continuous = false;

        recognition.interimResults = true;


        recognition.onstart = function() {

          isRecording = true;

          document.getElementById('mic-label').innerText = "Listening...";

          document.getElementById('btn-mic').classList.add('bg-rose-500/20', 'text-rose-400', 'border-rose-500/30');

        };


        recognition.onresult = function(event) {

          let transcript = '';

          for (let i = event.resultIndex; i < event.results.length; ++i) {

            transcript += event.results[i][0].transcript;

          }

          document.getElementById('recite-transcription').innerText = transcript;

        };


        recognition.onerror = function() {

          stopRecording();

        };


        recognition.onend = function() {

          stopRecording();

        };

      } else {

        document.getElementById('btn-mic').style.display = 'none';

      }

    }


    function toggleSpeechRecognition() {

      if (!recognition) return;

      if (isRecording) {

        recognition.stop();

      } else {

        recognition.start();

      }

    }


    function stopRecording() {

      isRecording = false;

      document.getElementById('mic-label').innerText = "Record Voice";

      document.getElementById('btn-mic').classList.remove('bg-rose-500/20', 'text-rose-400', 'border-rose-500/30');

    }


    // Stats Calculation

    function updateMasteryStats() {

      const pct = Math.round((masteredCards.size / SCRIPT_DATA.length) * 100);

      document.getElementById('header-mastery').innerText = `${pct}%`;

      document.getElementById('header-streak').innerText = userStreak;

    }
